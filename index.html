Initialisatie van het Radarscherm (Gefocust op Haven van Scheveningen) const map = L.map('radar-screen', { zoomControl: true, minZoom: 11, maxZoom: 17 }).setView([52.115, 4.240], 13); // Laad een native donkere navigatiekaart (CartoDB Dark Matter) L.tileLayer('https://{s}://{z}/{x}/{y}{r}.png', { attribution: '© OpenStreetMap © CARTO' }).addTo(map); // 2. Havenmonding & Havenhoofden intekenen (Scheveningen) const havenHoofden = [ [52.1023, 4.2612], // Noordelijk Havenhoofd [52.0998, 4.2635] // Zuidelijk Havenhoofd ]; L.polyline(havenHoofden, {color: '#00ff66', weight: 4, dashArray: '5, 5'}).addTo(map) .bindTooltip("Havenmond Scheveningen", {className: 'ship-radar-label'}); // Ankergebied 4 (Vóór de kust van Scheveningen) const ankerGebied4 = [ [52.140, 4.160], [52.160, 4.190], [52.145, 4.220], [52.125, 4.190] ]; L.polygon(ankerGebied4, {color: '#00ffff', weight: 1.5, fillColor: '#00ffff', fillOpacity: 0.03}).addTo(map) .bindTooltip("ANKERGEBIED 4", {permanent: true, className: 'ship-radar-label', direction: 'center'}); // 3. Database met actieve schepen (Start met 2 targets) let ships = [ { id: 1, name: "SCH-118 ZEEAREND", lat: 52.105, lon: 4.252, heading: 240, speed: 7.0, type: "Visserij (Trawler)", vectorLine: null, textLabel: null, marker: null }, { id: 2, name: "PACIFIC ROCK", lat: 52.135, lon: 4.185, heading: 135, speed: 0.0, type: "Bulk Carrier (Voor Anker)", vectorLine: null, textLabel: null, marker: null } ]; let selectedShipId = null; let namesPool = ["STORK 4", "EEMS TRADER", "ROTTERDAM STAR", "ZEELAND", "MIRFAK", "TRITON", "SCH-45", "NAVIGATOR"]; // 4. Update en Teken Targets + 6-Minuten Vectoren function renderShips() { ships.forEach(ship => { // Verwijder oude kaartlagen als ze bestaan if (ship.marker) map.removeLayer(ship.marker); if (ship.vectorLine) map.removeLayer(ship.vectorLine); if (ship.textLabel) map.removeLayer(ship.textLabel); // Bereken VTS Vector (Voorspelde positie over 6 minuten) const vectorMinutes = 6; const speedInDeg = (ship.speed * 0.000514) / 111320; // SOG naar graden const rad = (ship.heading * Math.PI) / 180; // Versnellingsfactor (x12) om beweging op het scherm zichtbaar te houden const endLat = ship.lat + Math.sin(rad) * speedInDeg * (vectorMinutes * 60) * 12; const endLon = ship.lon + Math.cos(rad) * speedInDeg * (vectorMinutes * 60) * 12; // Teken Koersvector (Lijn vooruit) ship.vectorLine = L.polyline([[ship.lat, ship.lon], [endLat, endLon]], {color: '#00ff66', weight: 1.5}).addTo(map); // Teken Radar Target Blip ship.marker = L.circleMarker([ship.lat, ship.lon], { radius: 5, color: '#00ff66', fillColor: '#04080f', fillOpacity: 1, weight: 2 }).addTo(map); // Teken VTS Label (Naam + Koers/Snelheid) ship.textLabel = L.marker([ship.lat + 0.0015, ship.lon + 0.002], { icon: L.divIcon({ className: 'ship-radar-label', html: `${ship.name}
${String(Math.round(ship.heading)).padStart(3,'0')}° / ${ship.speed.toFixed(1)}KT` }) }).addTo(map); // Selectie trigger bij klikken ship.marker.on('click', () => selectShip(ship.id)); }); } function selectShip(id) { selectedShipId = id; const ship = ships.find(s => s.id === id); if (!ship) return; document.getElementById('target-data').innerHTML = `
TARGET:${ship.name} TYPE:${ship.type} COG:${Math.round(ship.heading)}° True SOG:${ship.speed.toFixed(1)} Knopen POS:${ship.lat.toFixed(4)}N / ${ship.lon.toFixed(4)}E
`; calculateCPA(ship); } // 5. CPA (Closest Point of Approach) Gevaar Check function calculateCPA(selectedShip) { let warningBox = document.getElementById('cpa-warning-box'); warningBox.innerHTML = ""; ships.forEach(other => { if (other.id === selectedShip.id) return; const latDiff = selectedShip.lat - other.lat; const lonDiff = selectedShip.lon - other.lon; const distanceNM = Math.sqrt(latDiff * latDiff + lonDiff * lonDiff) * 60; // Afstand in zeemijlen const kabels = distanceNM * 10; // Alarm trigger als schepen binnen 4 kabels (800 meter) van elkaar varen en in beweging zijn if (distanceNM < 0.4 && (selectedShip.speed > 0 || other.speed > 0)) { warningBox.innerHTML += `
CPA RISK: ${other.name} (${kabels.toFixed(1)} Kabels)
`; } }); } // 6. EBL / VRM Tool (Peiling & Afstand Meten) let eblMode = false; let startPoint = null; let measurementLine = null; document.getElementById('tool-ebl-vrm').addEventListener('click', function() { eblMode = !eblMode; startPoint = null; if (measurementLine) map.removeLayer(measurementLine); if (eblMode) { this.classList.add('active-tool'); document.getElementById('measurement-output').innerText = "[EBL/VRM] Klik op de kaart voor het startpunt..."; } else { this.classList.remove('active-tool'); document.getElementById('measurement-output').innerText = "EBL/VRM inactief."; } }); map.on('click', (e) => { if (!eblMode) return; if (!startPoint) { startPoint = e.latlng; document.getElementById('measurement-output').innerText = "[EBL/VRM] Startpunt vastgezet. Klik nu op je doel..."; } else { const endPoint = e.latlng; if (measurementLine) map.removeLayer(measurementLine); // Teken stippellijn tussen punten measurementLine = L.polyline([startPoint, endPoint], {color: '#ffaa00', weight: 1.5, dashArray: '5, 5'}).addTo(map); // Bereken Ware Peiling (True Bearing) const lat1 = startPoint.lat * Math.PI / 180; const lat2 = endPoint.lat * Math.PI / 180; const dLon = (endPoint.lng - startPoint.lng) * Math.PI / 180; const y = Math.sin(dLon) * Math.cos(lat2); const x = Math.cos(lat1) * Math.sin(lat2) - Math.sin(lat1) * Math.cos(lat2) * Math.cos(dLon); let bearing = Math.atan2(y, x) * 180 / Math.PI; bearing = (bearing + 360) % 360; // Bereken Afstanden const meters = startPoint.distanceTo(endPoint); const nm = meters / 1852; const kabels = nm * 10; document.getElementById('measurement-output').innerHTML = ` PEILING (EBL): ${bearing.toFixed(0)}° True
AFSTAND (VRM):
• ${meters.toFixed(0)} meter
• ${nm.toFixed(2)} NM (Mijlen)
• ${kabels.toFixed(1)} Kabels `; // Reset tool na succesvolle meting eblMode = false; document.getElementById('tool-ebl-vrm').classList.remove('active-tool'); } }); // 7. Target Inzetten op Vaste Routes document.getElementById('spawn-btn').addEventListener('click', () => { if (ships.length >= 10) { alert("Maximum van 10 schepen in de sector bereikt!"); return; } const routeType = document.getElementById('spawn-route').value; const randomName = namesPool[Math.floor(Math.random() * namesPool.length)] + " " + Math.floor(Math.random() * 80 + 10); let newShip = { id: Date.now(), name: randomName, type: "Commercial Vessel" }; if (routeType === "inbound") { newShip.lat = 52.145; newShip.lon = 4.180; newShip.heading = 135; newShip.speed = 11.5; } else if (routeType === "outbound") { newShip.lat = 52.101; newShip.lon = 4.262; newShip.heading = 315; newShip.speed = 8.0; } else { newShip.lat = 52.125; newShip.lon = 4.225; newShip.heading = 345; newShip.speed = 5.0; } ships.push(newShip); renderShips(); logMessage("SYSTEM", `Nieuw target gedetecteerd: ${randomName}`, "system-msg"); }); // 8. VHF Marifoon Commando's (Koers & Snelheid aanpassen) document.getElementById('send-btn').addEventListener('click', handleVTSCommand); document.getElementById('vhf-input').addEventListener('keypress', (e) => { if (e.key === 'Enter') handleVTSCommand(); }); function handleVTSCommand() { const input = document.getElementById('vhf-input'); const text = input.value.trim().toLowerCase(); if (!text) return; if (!selectedShipId) { logMessage("SYSTEM", "Fout: Selecteer eerst een schip via de radar.", "system-msg"); input.value = ""; return; } const ship = ships.find(s => s.id === selectedShipId); logMessage("VTS SCHEVENINGEN", `${ship.name}, ${input.value}`, "vts-msg"); input.value = ""; setTimeout(() => { if (text.includes("koers")) { const num = text.match(/\d+/); if (num) { ship.heading = parseInt(num) % 360; logMessage(ship.name, `Begrepen VTS, sturen nu koers ${ship.heading}°`, "ship-msg"); } } else if (text.includes("snelheid")) { const num = text.match(/\d+/); if (num) { ship.speed = parseFloat(num); logMessage(ship.name, `Begrepen VTS, we passen snelheid aan naar ${ship.speed} knopen.`, "ship-msg"); } } else { logMessage(ship.name, "Herhaal aub? Gebruik 'koers [getal]' of 'snelheid [getal]'.", "ship-msg"); } renderShips(); }, 1000); } function logMessage(sender, text, type) { const log = document.getElementById('vhf-log'); log.innerHTML += `

${sender}: ${text}

`; log.scrollTop = log.scrollHeight; }
