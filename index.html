CPA (Closest Point of Approach) Berekening function calculateCPA(selectedShip) { let warningBox = document.getElementById('cpa-warning-box'); warningBox.innerHTML = ""; ships.forEach(other => { if (other.id === selectedShip.id) return; // Simpele afstandsberekening tussen twee schepen const latDiff = selectedShip.lat - other.lat; const lonDiff = selectedShip.lon - other.lon; const distanceNM = Math.sqrt(latDiff*latDiff + lonDiff*lonDiff) * 60; // Ruwe Zeemijlen if (distanceNM < 0.5 && selectedShip.speed > 0 && other.speed > 0) { warningBox.innerHTML += `
CPA RISK: ${other.name} (${(distanceNM * 10).toFixed(1)} Kabels)
`; } }); } // 6.
